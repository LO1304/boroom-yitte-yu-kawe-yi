3. Commandes de Lancement, Débogage & Installation APK

# 1. Lancer l'application Flutter réelle en direct (Edge Web) :
.\scripts\launch_preview.ps1 -Target edge -Port 8080

# 2. Débogage interactif avec Hot Reload [r], Hot Restart [R], DevTools [v] :
.\scripts\debug_preview.ps1 -Target edge -Port 8080

# 3. Installer l'APK Debug sur téléphone Android physique (USB Debugging activé) :
C:\Android\sdk\platform-tools\adb.exe devices
C:\Android\sdk\platform-tools\adb.exe install -r "mobile\build\app\outputs\flutter-apk\app-debug.apk"

# 4. Vérifications automatisées complètes :
node supabase\tests\validate_all.mjs
cd mobile && C:\flutter\bin\flutter.bat analyze && C:\flutter\bin\flutter.bat test
